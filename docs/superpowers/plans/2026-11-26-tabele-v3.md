# Tabele v3 — plan wdrożenia

> **For agentic workers:** REQUIRED SUB-SKILL: superpowers:subagent-driven-development. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Strony z tabelą wyglądają i działają jak makieta `S2Table`: widoki w zakładkach, pasek narzędzi (Tabela/Tablica, Grupuj, Filtruj, sort, Pola, szukajka, menu ⋯), grupy zwijane, zaznaczanie wielu wierszy, tablica kanban, import i eksport CSV, szablony po polsku z gotowymi widokami, nowa pusta strona i okno nowej strony.

**Architecture:** Rozbudowa obecnego `PageTable` + `use-table-store`. Widoki i grupowanie żyją w JSONB `content.settings` bloku (bez migracji schematu). Czyste funkcje (widoki, grupy, filtry, CSV) w modelu, UI w osobnych komponentach w `src/features/pages/table/`. Jedna migracja danych: polskie tytuły stron z domyślną angielską nazwą.

**Tech Stack:** Next.js 16 (cacheComponents), React 19, Supabase, Tailwind 4, `@/components/controls`, `lucide-react` 1.73, Vitest, Playwright i pgTAP (tylko CI).

**Spec:** `docs/superpowers/specs/2026-11-26-tabele-v3-design.md` — przeczytaj w całości przed każdym zadaniem.

**Makiety (wiążące, wygrywają z planem):** `docs/design/mockups/S2Table.dc.html` (logika i dane wariantów w `<script type="text/x-dc">`, sekcja `K = { projects, books, subs, passwords, blank, websites }`), warianty `S2Books`, `S2Main`, `S2Projects`, `S2Websites`, `S2Passwords`, `S2Vault*`, `S2TableMenu`, `S2ColumnMenu`, `S2FilterPopover`, `S2CardDialog`, `S2CardMenu`, `S2AddField`, `S2PageMenu`, `S2SidebarCollapsed`, a także `S2EmptyPage`, `S2NewPage`, `S2ImportCSV`, `S2ImportCSVMap`, `S2Sidebar`. Zrzut: `PW_CHANNEL=chrome pnpm -s mockups:shots <Nazwa>` → `test-results/mockups/<Nazwa>.png`. Interakcje (otwarte menu, tablica, zaznaczenie) sprawdzaj, klikając makietę w przeglądarce (Playwright z `channel: 'chrome'`).

### Kontrakty wspólne

```ts
// src/features/pages/table-model.ts — rozszerzenia
// FilterCond["op"] dostaje: "within_days" (value: number, data od dziś do dziś+N) | "this_year" (rok daty = rok dziś)
//                        | "not_in" (wybór: wartość spoza listy) | "not_equals" (tekst różny od value, bez wielkości liter)
export type TableView = {
  id: string;               // losowe, np. crypto.randomUUID()
  name: string;             // 1–40 znaków
  filters: FilterCond[];
  sort: SortKey[];
  group_by: string | null;  // id pola select/status/checkbox; null = bez grup
  layout: "table" | "board";
  hidden_field_ids: string[];
};
// TableSettings dostaje:
//   views: TableView[];                 // zapisane widoki poza „Wszystkie”
//   active_view_id: string | null;      // null = „Wszystkie”
//   group_by: string | null;            // grupowanie „Wszystkich”
//   layout: "table" | "board";          // układ „Wszystkich”
//   collapsed_groups: Record<string, string[]>; // id widoku albo "all" → klucze zwiniętych grup

// src/features/pages/table-views.ts (nowy, czyste funkcje)
export type EffectiveView = Pick<TableView, "filters" | "sort" | "group_by" | "layout" | "hidden_field_ids">;
export function effectiveView(settings: TableSettings): EffectiveView;           // aktywny widok albo „Wszystkie”
export function searchRows(fields: PageField[], rows: PageRow[], q: string): PageRow[]; // bez pól secret, bez wielkości liter i polskich znaków diakrytycznych
export type RowGroup = { key: string; label: string; color: string | null; rows: PageRow[] }; // key "" = „Bez wartości”
export function groupRows(fields: PageField[], rows: PageRow[], groupBy: string | null): RowGroup[]; // kolejność opcji pola, puste na końcu
export function viewCount(fields: PageField[], rows: PageRow[], view: EffectiveView, today: string): number;

// src/features/pages/csv.ts (nowy, czyste funkcje)
export function toCsv(fields: PageField[], rows: PageRow[]): string;   // RFC 4180, separator „,”, BOM UTF-8, nagłówek = nazwy pól, pola secret pominięte, wartości jak formatValue (daty RRRR-MM-DD, liczby z kropką)
export function parseCsv(text: string): { header: string[]; rows: string[][] }; // cudzysłowy, "" w polu, CRLF/LF, BOM, separator „,” albo „;” (wykryty z nagłówka)
export type CsvMapping = { column: number; target: { kind: "field"; field_id: string } | { kind: "new"; name: string; type: FieldType } | { kind: "skip" } }[];
export function suggestMapping(header: string[], fields: PageField[]): CsvMapping; // dopasowanie po nazwie bez wielkości liter; reszta "new" typu text
export function csvToValues(rows: string[][], mapping: CsvMapping, fields: PageField[]): { rows: Record<string, unknown>[]; newOptions: Record<string, FieldOption[]> }; // klucze wartości: id pola albo "new:<indeks>" (nowe pola w kolejności mapowania); konwersja przez normalizeValue; opcje select dopasowane po etykiecie, nieznane → nowa opcja (id nadaje funkcja) zwracana w newOptions pod tym samym kluczem; pola secret pomijane
export const CSV_MAX_BYTES = 1_048_576;

// src/features/pages/templates.ts — rozszerzenia
// TemplateConfig dostaje: newLabel: string; noun: [string, string, string];
//   views: { name: string; filters: { field: string; op: FilterCond["op"]; value?: unknown; option?: string[] }[] }[];
//   group: string | null; board: string | null; collapsed: string[]  (klucze opcji)
export const LEGACY_TEMPLATE_TITLES: Record<"books" | "projects" | "websites" | "subscriptions", string>; // stare angielskie nazwy
export function templateViews(key: TemplateKey, fields: PageField[]): TableView[]; // widoki szablonu z prawdziwymi id pól/opcji

// src/features/pages/actions.ts — nowe akcje (wzorem addRow/deleteRow: auth, walidacja zod, user_id)
export async function deleteRows(ids: string[]): Promise<{ ok: true } | Blad>;        // miękkie usunięcie (deleted_at), max 500
export async function restoreRows(ids: string[]): Promise<{ ok: true } | Blad>;
export async function duplicateRows(ids: string[]): Promise<{ ok: true; rows: { id: string; values: Record<string, unknown>; position: number }[] } | Blad>;
export async function setRowsValue(ids: string[], fieldId: string, value: unknown): Promise<{ ok: true } | Blad>;
export async function importCsvRows(blockId: string, newFields: { name: string; type: FieldType }[], rows: Record<string, unknown>[], newOptions?: Record<string, FieldOption[]>): Promise<{ ok: true; fieldIds: string[]; count: number } | Blad>; // max 2000 wierszy; w rows klucze "new:<indeks>" wskazują nowe pola; newOptions: klucz = id pola wyboru albo "new:<indeks>"; odmowa dla bloku strony is_system lub bloku/strony w koszu; przy błędzie sprząta po sobie (best effort)
```
